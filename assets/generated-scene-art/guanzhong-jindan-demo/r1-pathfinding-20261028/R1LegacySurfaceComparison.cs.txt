using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Reflection;
using NUnit.Framework;
using TianZhang.Features.CombatPresentation;
using UnityEngine;

namespace TianZhang.Tests.PlayMode
{
    internal static class R1LegacySurfaceComparison
    {
        internal static void Run(CombatBattlefieldSurface surface,Static3DCombatUnitPresentationProvider provider,string output)
        {
            var assembly=typeof(CombatBattlefieldSurface).Assembly;
            Type current=assembly.GetType("TianZhang.Features.CombatPresentation.CombatUnitSurfacePose");
            Type legacy=assembly.GetType("TianZhang.Features.CombatPresentation.R1LegacyCombatUnitSurfacePose");
            var rows=new List<string>{"unit,x,z,yaw,position_error,rotation_error,status"};
            int compared=0;
            foreach(string id in new[] {"player","enemy","enemy_2"})
            {
                Assert.IsTrue(provider.TryGetPresentedUnit(id,out var root));
                Assert.IsTrue(provider.TryGetPresentedPose(id,out var pose));
                object Create(Type type)=>Activator.CreateInstance(type,BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic,null,
                    new object[] {surface,root.transform.GetChild(0),pose.IsDoubleCell},null);
                object before=Create(legacy),after=Create(current);
                foreach(var hex in new[] {new Vector2(2,2),new Vector2(2,3),new Vector2(2,4),new Vector2(2,5),new Vector2(2,6),
                    new Vector2(4,7),new Vector2(8,8),new Vector2(8,9),new Vector2(8,10),new Vector2(5,11),new Vector2(5,12),new Vector2(5,13),new Vector2(4,0)})
                foreach(float offset in new[] {0f,.3f}) foreach(float yaw in new[] {0f,60f,120f,180f,240f,300f})
                {
                    Vector3 center=CombatBattlefieldSurface.Horizontal((int)hex.x,(int)hex.y)+Vector3.forward*offset;
                    object[] a={center,yaw,null,null},b={center,yaw,null,null};
                    string Failure(Type t,object instance,object[] args)
                    {try {t.GetMethod("Evaluate").Invoke(instance,args);return null;}catch(TargetInvocationException e){return e.InnerException.Message;}}
                    string oldReason=Failure(legacy,before,a),newReason=Failure(current,after,b);
                    Assert.AreEqual(oldReason,newReason);
                    if(oldReason!=null){rows.Add(id+","+center.x+","+center.z+","+yaw+",0,0,"+oldReason);continue;}
                    float pe=Vector3.Distance((Vector3)a[2],(Vector3)b[2]);
                    Quaternion ar=(Quaternion)a[3],br=(Quaternion)b[3];
                    float re=Mathf.Max(Mathf.Abs(ar.x-br.x),Mathf.Abs(ar.y-br.y),Mathf.Abs(ar.z-br.z),Mathf.Abs(ar.w-br.w));
                    Assert.LessOrEqual(pe,.000001f,id+" position "+center); Assert.LessOrEqual(re,.000001f,id+" rotation "+center);
                    rows.Add(string.Join(",",id,center.x,center.z,yaw,pe.ToString("G9",CultureInfo.InvariantCulture),re.ToString("G9",CultureInfo.InvariantCulture),"matched"));
                    compared++;
                }
            }
            Assert.Greater(compared,100);File.WriteAllLines(Path.Combine(output,"legacy-surface.csv"),rows);
        }
    }
}
