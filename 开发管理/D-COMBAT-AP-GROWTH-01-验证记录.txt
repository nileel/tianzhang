```python
from fractions import Fraction as F
base = [2, 4, 12, 48, 240, 1440]
shares = [F(30, 100), F(32, 100), F(38, 100)]
for realm, k in [(1, 1), (1, 9), (2, 2), (3, 3), (5, 4)]:
    prior = sum(F(b, 4) for b in base[1:realm])
    portion = F(k, 9) if realm == 1 else sum(shares[:k]) if realm in (2, 3) else F(k, 4)
    v = F(base[realm]) + prior + F(base[realm], 4) * portion
    milli = (v * 1000).numerator // (v * 1000).denominator
    print(realm, k, milli, max(1, 100000000 // milli))
```
